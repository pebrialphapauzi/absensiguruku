String(val) : ''; }); return obj; }); } // ========================================== // 1. DASHBOARD & REKAP DATA UTAMA // ========================================== function getDashboardData(tanggalFilter) { try { const siswa = getSheetDataObjects(SHEET_NAMES.SISWA); const kelas = getSheetDataObjects(SHEET_NAMES.KELAS); const absensi = getSheetDataObjects(SHEET_NAMES.ABSENSI); const materi = getSheetDataObjects(SHEET_NAMES.MATERI); const targetDate = tanggalFilter || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); const absensiHariIni = absensi.filter(a => a.Tanggal === targetDate); let hadir = 0, izin = 0, sakit = 0, alfa = 0; absensiHariIni.forEach(item => { const st = String(item.Status).toLowerCase(); if (st === 'hadir') hadir++; else if (st === 'izin') izin++; else if (st === 'sakit') sakit++; else if (st === 'alfa') alfa++; }); const totalSiswa = siswa.length; const persentaseKehadiran = totalSiswa > 0 ? Number(((hadir / totalSiswa) * 100).toFixed(1)) : 0; return { success: true, tanggal: targetDate, stats: { totalSiswa, totalKelas: kelas.length, hadir, izin, sakit, alfa, persentaseKehadiran }, siswa, kelas, absensi, materi }; } catch (err) { return { success: false, message: 'Gagal memuat data: ' + err.message }; } } // ========================================== // 3. MANAJEMEN DATA SISWA & IMPORT EXCEL // ========================================== function simpanSiswa(data, isEdit) { try { setupDatabase(); const sheet = getSpreadsheet().getSheetByName(SHEET_NAMES.SISWA); const list = getSheetDataObjects(SHEET_NAMES.SISWA); const nisClean = String(data.nis || '').trim(); const namaClean = String(data.namaLengkap || '').trim(); const jkClean = String(data.jenisKelamin || 'L').trim(); const kelasClean = String(data.kelas || '').trim(); if (!nisClean || !namaClean || !kelasClean) { return { success: false, message: 'NIS, Nama Lengkap, dan Kelas wajib diisi.' }; } const qrString = 'INFORQR-' + nisClean; if (isEdit) { const existing = list.find(s => String(s.NIS).trim() === nisClean); if (!existing) return { success: false, message: 'Siswa dengan NIS tersebut tidak ditemukan.' }; const r = existing._rowIndex; sheet.getRange(r, 2).setValue(namaClean); sheet.getRange(r, 3).setValue(jkClean); sheet.getRange(r, 4).setValue(kelasClean); sheet.getRange(r, 5).setValue(qrString); return { success: true, message: 'Data siswa berhasil diperbarui.' }; } else { const duplicate = list.find(s => String(s.NIS).trim() === nisClean); if (duplicate) { return { success: false, message: 'NIS ' + nisClean + ' sudah terdaftar.' }; } sheet.appendRow([nisClean, namaClean, jkClean, kelasClean, qrString]); return { success: true, message: 'Siswa berhasil ditambahkan beserta QR Code unik.' }; } } catch (err) { return { success: false, message: 'Gagal menyimpan siswa: ' + err.message }; } } function hapusSiswa(nis) { try { const sheet = getSpreadsheet().getSheetByName(SHEET_NAMES.SISWA); const list = getSheetDataObjects(SHEET_NAMES.SISWA); const target = list.find(s => String(s.NIS).trim() === String(nis).trim()); if (!target) return { success: false, message: 'Data siswa tidak ditemukan.' }; sheet.deleteRow(target._rowIndex); return { success: true, message: 'Data siswa berhasil dihapus.' }; } catch (err) { return { success: false, message: 'Gagal menghapus siswa: ' + err.message }; } } function importSiswaBulk(rowsArray) { try { setupDatabase(); const siswaList = getSheetDataObjects(SHEET_NAMES.SISWA); const absensiList = getSheetDataObjects(SHEET_NAMES.ABSENSI); const rawCode = String(payload.qrCode || '').trim(); const nisFromQr = rawCode.startsWith('INFORQR-') ? rawCode.replace('INFORQR-', '').trim() : rawCode; } if (payload.kelasSesi && String(siswa.Kelas).trim() !== String(payload.kelasSesi).trim()) { return { success: false, statusCode: 'WRONG_CLASS', message: `${siswa.Nama_Lengkap} terdaftar di kelas ${siswa.Kelas}, bukan ${payload.kelasSesi}.` }; } const now = new Date(); const tanggal = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd'); const jam = Utilities.formatDate(now, Session.getScriptTimeZone(), 'HH:mm:ss'); const sesiId = String(payload.sesiId || tanggal); return { success: true, message: `Absensi berhasil: ${siswa.Nama_Lengkap} (${siswa.Kelas}) - HADIR`, record: { ID_Absensi: idAbsensi, Tanggal: tanggal, Jam: jam, NIS: siswa.NIS, Nama_Lengkap: siswa.Nama_Lengkap, Kelas: siswa.Kelas, Mata_Pelajaran: mapel, Status: 'Hadir', Keterangan: 'Scan QR Code', Sesi_ID: sesiId } }; } catch (err) { return { success: false, message: 'Gagal memproses absensi QR: ' + err.message }; } } function simpanStatusManual(payload) { try { setupDatabase(); const sheet = getSpreadsheet().getSheetByName(SHEET_NAMES.ABSENSI); const jam = Utilities.formatDate(now, Session.getScriptTimeZone(), 'HH:mm:ss'); const idAbsensi = 'ABS-' + now.getTime(); sheet.appendRow([ idAbsensi, tanggal, jam, siswa.NIS, siswa.Nama_Lengkap, siswa.Kelas, payload.mataPelajaran || 'Informatika', statusClean, ketClean || 'Input manual guru', payload.sesiId || tanggal ]);
